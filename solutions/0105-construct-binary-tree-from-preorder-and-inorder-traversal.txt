        TreeNode* root = new TreeNode(root_val);
        
        int mid = inorder_map[root_val];
        int left_size = mid - in_left;
        
        root->left = helper(preorder, pre_left + 1, pre_left + left_size, 
        in_left, mid - 1);
        root->right = helper(preorder, pre_left + left_size + 1, pre_right, 
        mid + 1, in_right);
        
        return root;
    }
};
