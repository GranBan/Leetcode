    void pathSumBuilder(TreeNode* root, std::vector<int>& temp, int& target) {
        if (root == nullptr) return;
        temp.push_back(root->val);
        if (root->left == nullptr && root->right == nullptr && target - 
        root->val == 0) {
            tree_res.push_back(temp);
        } else {
            target -= root->val;
            pathSumBuilder(root->left, temp, target);
            pathSumBuilder(root->right, temp, target);
            target += root->val;
        }
        temp.pop_back();
    }
};
